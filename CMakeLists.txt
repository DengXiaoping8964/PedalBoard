cmake_minimum_required(VERSION 3.22)

project(PedalBoard
        VERSION 1.0.0
        DESCRIPTION "Guitar pedalboard with physical monitoring and VB-CABLE virtual routing"
        LANGUAGES C CXX)

#==============================================================================
# PedalBoard is a Windows-only application. It relies on WASAPI device
# enumeration and on the VB-CABLE driver being installed by the user.
#==============================================================================
if (NOT WIN32)
    message(FATAL_ERROR "PedalBoard targets Windows only (WASAPI/ASIO + VB-CABLE routing).")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT PedalBoard)

#==============================================================================
# JUCE 8 via FetchContent.
# The tag is pinned so that the DSP/device APIs used here are the ones this
# source tree was written against. Override with
#   -DPEDALBOARD_JUCE_TAG=8.0.x
# if you need another 8.x release.
#==============================================================================
set(PEDALBOARD_JUCE_TAG "8.0.15" CACHE STRING "JUCE git tag to build against")
set(PEDALBOARD_JUCE_LOCAL "" CACHE STRING "Path to an existing JUCE checkout (skips FetchContent)")

include(FetchContent)

if (PEDALBOARD_JUCE_LOCAL)
    message(STATUS "PedalBoard: using local JUCE at ${PEDALBOARD_JUCE_LOCAL}")
    add_subdirectory("${PEDALBOARD_JUCE_LOCAL}" "${CMAKE_CURRENT_BINARY_DIR}/_juce" EXCLUDE_FROM_ALL)
else()
    # GIT_SHALLOW needs a tag/branch, which is why this is a tag and not a commit.
    FetchContent_Declare(JUCE
                        GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
                        GIT_TAG        ${PEDALBOARD_JUCE_TAG}
                        GIT_SHALLOW    TRUE
                        GIT_PROGRESS   TRUE)
    FetchContent_MakeAvailable(JUCE)
endif()

#==============================================================================
# Steinberg ASIO SDK is NOT redistributable. Drop it into ThirdParty/asio and
# configure with -DPEDALBOARD_ASIO=ON to get ASIO device support; otherwise the
# app runs on WASAPI (shared/exclusive) and MME.
#==============================================================================
option(PEDALBOARD_ASIO "Build with ASIO support (requires a locally licensed Steinberg ASIO SDK)" OFF)
set(PEDALBOARD_ASIO_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/ThirdParty/asio" CACHE PATH "ASIO SDK root")

if (PEDALBOARD_ASIO AND NOT EXISTS "${PEDALBOARD_ASIO_SDK_DIR}/common/asiosys.h")
    message(FATAL_ERROR "PEDALBOARD_ASIO=ON but no ASIO SDK found at ${PEDALBOARD_ASIO_SDK_DIR}")
endif()

#==============================================================================
# Impulse responses, embedded so the cabinet simulator has three IRs available
# with no external download. These are generated by Resources/make_irs.mjs
# (run `node Resources/make_irs.mjs` to regenerate). The script lives outside
# Resources/IRs because that folder is globbed by juce_add_binary_data, and a
# .mjs in there would be compiled into the executable as data.
#==============================================================================
set(PEDALBOARD_IR_FILES
    "${CMAKE_CURRENT_SOURCE_DIR}/Resources/IRs/ir_1x12_v30_closed.wav"
    "${CMAKE_CURRENT_SOURCE_DIR}/Resources/IRs/ir_2x12_open_back.wav"
    "${CMAKE_CURRENT_SOURCE_DIR}/Resources/IRs/ir_4x12_high_gain.wav")

foreach (_ir IN LISTS PEDALBOARD_IR_FILES)
    if (NOT EXISTS "${_ir}")
        message(FATAL_ERROR
            "Missing impulse response ${_ir}\n"
            "Generate them first:  node Resources/make_irs.mjs")
    endif()
endforeach()

# juce_add_binary_data takes SOURCES, not a directory: in 8.0.15 the function
# parses only NAMESPACE, HEADER_NAME and SOURCES, so a SOURCE_DIR keyword would be
# silently dropped and the call would fail with "must be passed at least one file to
# encode". The list is spelled out to match the FATAL_ERROR check above exactly.
juce_add_binary_data(PedalBoardIRs
                     SOURCES      ${PEDALBOARD_IR_FILES}
                     HEADER_NAME  "PedalBoardIRData.h"
                     NAMESPACE    PedalBoardIR)

#==============================================================================
# The application target.
#
# Only the keywords _juce_initialise_target actually parses are passed: it takes
# PRODUCT_NAME, COMPANY_NAME, COMPANY_WEBSITE and COMPANY_COPYRIGHT among others,
# and cmake_parse_arguments silently drops anything else - so BINARY_NAME and
# COMPANY_WEBSITE_URL here would not be an error, just dead text that looks like
# it configured something. The output name comes from PRODUCT_NAME.
#==============================================================================
juce_add_gui_app(PedalBoard
                 PRODUCT_NAME       "PedalBoard"
                 COMPANY_NAME       "PedalBoard"
                 COMPANY_WEBSITE    "https://vb-audio.com/Cable/"
                 COMPANY_COPYRIGHT  "PedalBoard. Uses VB-CABLE, Donationware by Vincent Burel, which this app does not bundle or install.")

target_sources(PedalBoard PRIVATE
    Source/Main.cpp

    Source/Audio/AudioEngine.cpp

    Source/Common/PedalLanguage.cpp

    Source/Effects/EffectFactory.cpp

    Source/VirtualDevice/VBCableRouter.cpp

    Source/UI/DarkLookAndFeel.cpp
    Source/UI/EffectCard.cpp
    Source/UI/DeviceChooserPanel.cpp
    Source/UI/MainComponent.cpp)

target_include_directories(PedalBoard PRIVATE
    Source
    "${CMAKE_CURRENT_SOURCE_DIR}/vendor/signalsmith-stretch/repo/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/vendor/signalsmith-stretch/linear/include")

set(PEDALBOARD_JUCE_FLAGS
    # --- audio back ends ------------------------------------------------------
    # Every name here was found in modules/juce_audio_devices/juce_audio_devices.h
    # or modules/juce_gui_extra/juce_gui_extra.h of the 8.0.15 checkout.
    JUCE_ASIO=${PEDALBOARD_ASIO}
    JUCE_WASAPI=1
    JUCE_WASAPI_EXCLUSIVE=1
    JUCE_DIRECTSOUND=1
    JUCE_ALSA=0
    JUCE_COREAUDIO=0
    JUCE_AUHAL=0
    JUCE_OBOE=0
    JUCE_USE_ANDROID_OPENSLES=0
    # --- things this app must not pull in -------------------------------------
    # No browser, no curl, no splash. These belong to juce_gui_extra, which is no
    # longer linked, so they are belt-and-braces: kept because a later module
    # change would otherwise silently re-enable WebView2, whose default in 8.0.15
    # is off but which JUCE_WEB_BROWSER=0 alone does not imply.
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_USE_WIN_WEBVIEW2=0
    JUCE_PUSH_NOTIFICATIONS=0
    JUCE_DISPLAY_SPLASH_SCREEN=0
    # --- audio codecs ---------------------------------------------------------
    # FLAC and Ogg are on by default; MP3 is not (JUCE leaves the patent-era
    # decoder off unless asked). The cabinet's custom-IR loader advertises every
    # format the built-in AudioFormatManager can actually decode, so the decoder
    # has to be compiled in rather than merely listed in a file dialog.
    JUCE_USE_MP3AUDIOFORMAT=1
    # --- plugin hosting -------------------------------------------------------
    # The Plugin Host card loads a .vst3 effect from disk. That is the only
    # plugin format implemented here, and it is compiled into JUCE (the VST3 SDK
    # lives in the JUCE tree), so nothing has to be downloaded separately.
    # Enabling this pulls juce_audio_processors and juce_gui_extra in with it.
    JUCE_PLUGINHOST_VST3=1
    JUCE_PLUGINHOST_VST=0
    JUCE_PLUGINHOST_AU=0
    JUCE_PLUGINHOST_LADSPA=0
    # --- core behaviour -------------------------------------------------------
    JUCE_STRICT_REFCOUNTEDPOINTER=1)

# Deliberately not set, because the build system already decides them and a second
# value on the command line is noise: JUCE_MODULE_AVAILABLE_* (derived from the
# modules linked below), JUCE_STANDALONE_APPLICATION (juce_add_gui_app defines it),
# and JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED.

target_compile_definitions(PedalBoard PRIVATE ${PEDALBOARD_JUCE_FLAGS})

if (PEDALBOARD_ASIO)
    target_include_directories(PedalBoard PRIVATE "${PEDALBOARD_ASIO_SDK_DIR}/common")
endif()

target_link_libraries(PedalBoard PRIVATE
    PedalBoardIRs
    # Only the modules whose headers are actually #included are listed. Each
    # juce_add_module target forwards its declared dependencies as INTERFACE
    # link libraries (JUCEModuleSupport.cmake: _juce_get_metadata ...
    # target_link_libraries(${module_name} INTERFACE ${module_dependencies})),
    # so juce_core, juce_events, juce_graphics, juce_data_structures and
    # juce_audio_formats - the last pulled by juce_dsp - arrive on their own.
    #
    # juce_audio_processors is here only for the Plugin Host effect: it owns the
    # VST3 format object and AudioPluginInstance. It also pulls juce_gui_extra,
    # which is what an AudioProcessorEditor needs, so that module is no longer
    # absent - WebView2 and curl stay off through the flags above.
    # juce_audio_utils stays absent: nothing here needs its AudioDeviceSelector.
    juce::juce_audio_basics
    juce::juce_audio_devices
    juce::juce_audio_processors
    juce::juce_core
    juce::juce_dsp
    juce::juce_gui_basics
    juce::juce_recommended_config_flags
    juce::juce_recommended_lto_flags
    juce::juce_recommended_warning_flags)

# Crash triage: /MAP writes a linker map next to the binary listing every function
# with its RVA, which turns a WER fault offset into a symbol without needing a
# debugger installed. Off by default because the map is large.
option(PEDALBOARD_EMIT_MAP "Ask the linker for a .map file (crash triage)" OFF)
if (PEDALBOARD_EMIT_MAP AND MSVC)
    target_link_options(PedalBoard PRIVATE "/MAP")
endif()

if (MSVC)
    # Only the options this project adds. /Ox and /EHsc come from
    # juce_recommended_config_flags and whole-program optimisation from
    # juce_recommended_lto_flags, so repeating them here would just be a second
    # opinion on the same command line.
    #
    # /utf-8 is not cosmetic: the Chinese string table in Source/Common/
    # PedalLanguage.cpp is a UTF-8 literal in a UTF-8 source file, and without
    # this MSVC decodes the literal in the system ANSI codepage, so the bytes
    # handed to the translator are wrong before the program ever runs.
    #
    # /bigobj because every effect lives in a header and PedalBoardIRData.cpp is
    # generated; /wd4324 silences the padding-structure warning the atomic pairs in
    # AudioRingBuffer provokes.
    target_compile_options(PedalBoard PRIVATE /utf-8 /Ob2 /fp:fast /bigobj /wd4324)
    target_compile_definitions(PedalBoard PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()

#==============================================================================
# Also drop the IRs next to the executable so users can inspect/replace them.
#==============================================================================
add_custom_command(TARGET PedalBoard POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:PedalBoard>/IRs"
    COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_CURRENT_SOURCE_DIR}/Resources/IRs"
            "$<TARGET_FILE_DIR:PedalBoard>/IRs"
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:PedalBoard>/Plugins"
    COMMENT "Copying impulse responses and creating the Plugins folder next to PedalBoard.exe")
